% Chapter 22, concept figure: an LP polyhedron, its lattice points, the LP
% optimum at a fractional vertex and the integer optimum elsewhere.
\begin{tikzpicture}[scale=0.95]
  % feasible polyhedron: 3x+2y<=15, -x+2y<=5, x<=4, x,y>=0
  \fill[sbBlue!15] (0,0) -- (4,0) -- (4,1.5) -- (2.5,3.75) -- (0,2.5) -- cycle;
  \draw[sbBlue!60] (1.5,5.25) -- (4.6,0.6);
  \draw[sbBlue!60] (-0.2,2.4) -- (4.6,4.8);
  \draw[sbBlue!60] (4,-0.2) -- (4,2.2);
  \draw[sbBlue,thick] (0,0) -- (4,0) -- (4,1.5) -- (2.5,3.75) -- (0,2.5) -- cycle;
  \node[sbannot,rotate=-56] at (4.05,2.95) {$3x+2y\le 15$};
  \node[sbannot,rotate=26.6] at (1.1,3.35) {$-x+2y\le 5$};
  \node[sbannot,right] at (4.05,0.9) {$x\le 4$};
  % axes
  \draw[->] (-0.3,0) -- (5.7,0) node[right,font=\small] {$x$};
  \draw[->] (0,-0.3) -- (0,5.5) node[above,font=\small] {$y$};
  \foreach \i in {1,...,5} {
    \draw (\i,0.08) -- (\i,-0.08) node[below,font=\scriptsize] {\i};
    \draw (0.08,\i) -- (-0.08,\i) node[left,font=\scriptsize] {\i};
  }
  % level lines of the objective x+2y
  \draw[dashed,sbOrange!70] (0,3) -- (5.6,0.2);
  \draw[dashed,sbOrange!70] (0,4.5) -- (5.6,1.7);
  \draw[dashed,sbOrange] (0,5) -- (5.6,2.2);
  \node[sbannot,text=sbOrange,rotate=-26.6,anchor=south] at (4.9,2.7) {$x+2y=10$};
  \node[sbannot,text=sbOrange,rotate=-26.6,anchor=south] at (4.9,2.2) {$x+2y=9$};
  \node[sbannot,text=sbOrange,rotate=-26.6,anchor=south] at (4.9,0.7) {$x+2y=6$};
  \draw[sbvec,draw=sbOrange] (0.7,0.5) -- ++(0.4,0.8);
  \node[sbannot,text=sbOrange,anchor=west] at (1.15,1.25) {$\vect{c}=(1,2)$};
  % lattice points inside the polyhedron
  \foreach \x/\ymax in {0/2,1/3,2/3,3/3,4/1} {
    \foreach \y in {0,...,\ymax} {\fill[black] (\x,\y) circle (1.7pt);}
  }
  % optima and the rounded points
  \node[circle,fill=sbOrange,inner sep=2.2pt] at (2.5,3.75) {};
  \node[sbannot,text=sbOrange,anchor=south west] at (2.6,3.9) {LP optimum $(2.5,3.75)$, value $10$};
  \node[circle,fill=sbGreen,inner sep=2.2pt] at (3,3) {};
  \node[sbannot,text=sbGreen,anchor=north west] at (3.12,2.9) {integer optimum $(3,3)$, value $9$};
  \node[text=sbRed,font=\small\bfseries] at (2,4) {$\times$};
  \node[text=sbRed,font=\small\bfseries] at (3,4) {$\times$};
  \node[sbannot,text=sbRed,anchor=east] at (1.8,4.15) {rounded: infeasible};
\end{tikzpicture}

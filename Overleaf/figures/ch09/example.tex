  \draw[sbpathalt,->] (2.35,0.5) -- (2.35,2.3);

  \node[sbannot,text=sbRed,align=center] at (2.5,3.45) {$a_2$ and $a_3$ both at $(2,1)$ at $t=1$;\\ $a_1$ arrives there at $t=2$};

% The two-level idea of CBS.  Left: planned alone, two shortest paths meet
% at cell v at time t.  Right: the high level splits the conflict into two
% constraint-tree children; each replans one agent with the low level.
\begin{tikzpicture}[font=\small]
% left: the conflict
\begin{scope}[scale=0.8]
  \drawgrid{5}{3}
  \fill[pattern=north east lines,pattern color=sbRed!80] (2,1) rectangle ++(1,1);
  \draw[sbpath,->] (0.5,1.5) -- (4.35,1.5);
  \draw[sbpathalt,->] (2.5,0.5) -- (2.5,2.35);
  \node[sbagentA] at (0.5,1.5) {1};
  \node[sbagentB] at (2.5,0.5) {2};
  \node[sbgoal,label={[sbannot]above:$\gamma_1$}] at (4.5,1.5) {};
  \node[sbgoal,label={[sbannot]right:$\gamma_2$}] at (2.5,2.5) {};
  \node[sbannot,text=sbRed] at (2.5,3.4) {both at $v$ at time $t$};
  \node[sbannot,text=black,align=center] at (2.5,-0.7) {each agent planned alone:\\ shortest paths that meet};
\end{scope}
% right: the constraint tree
\begin{scope}[xshift=6.3cm]
  \node[sbbox,minimum width=4.2cm] (root) at (2.4,2.6) {root: no constraints\\ conflict $\langle a_1,a_2,v,t\rangle$};
  \node[sbboxalt,minimum width=3.1cm] (l) at (0.3,0.5) {add $\langle a_1,v,t\rangle$\\ replan $a_1$ only};
  \node[sbboxalt,minimum width=3.1cm] (r) at (4.5,0.5) {add $\langle a_2,v,t\rangle$\\ replan $a_2$ only};
  \draw[sbflow] (root) -- (l);
  \draw[sbflow] (root) -- (r);
  \node[sbannot,align=center] at (2.4,-0.8) {high level: best-first search on the sum of costs\\ low level: space-time \astar for one agent under its constraints};
\end{scope}
\end{tikzpicture}

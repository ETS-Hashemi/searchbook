% Concept figure for Chapter 11: subdimensional expansion.  The joint space
% of two agents is drawn as a lattice; the individual policies trace a
% one-dimensional path through it, which M* follows until a collision forces
% it to search a small two-dimensional patch.
\begin{tikzpicture}[x=0.72cm,y=0.72cm,font=\small]
% the joint space: an 8 x 8 lattice of configurations
\foreach \i in {0,...,7} \foreach \j in {0,...,7} {\fill[black!22] (\i,\j) circle (1.6pt);}
% the patch that the search explores after the collision
\fill[sbOpen,opacity=0.75,rounded corners=4pt] (1.55,1.55) rectangle (4.45,4.45);
\draw[sbBlue!60,rounded corners=4pt] (1.55,1.55) rectangle (4.45,4.45);
% the policy path (diagonal) and the collision on it
\draw[sbpath] (0,0) -- (2,2);
\draw[sbpath,dashed,opacity=0.45] (2,2) -- (5,5);
\draw[sbpath] (5,5) -- (7,7);
\fill[sbRed] (3,3) circle (4.5pt);
\draw[sbRed,thick] (2.7,2.7) -- (3.3,3.3); \draw[sbRed,thick] (2.7,3.3) -- (3.3,2.7);
% the detour found inside the patch: agent 2 waits, agent 1 passes
\draw[sbpathalt] (2,2) -- (2,3) -- (3,4) -- (4,4) -- (5,5);
\foreach \i in {0,...,7} {\fill[sbBlue] (\i,\i) circle (2.2pt);}
\fill[sbOrange] (2,3) circle (2.2pt); \fill[sbOrange] (3,4) circle (2.2pt); \fill[sbOrange] (4,4) circle (2.2pt);
\node[sbstart,label={[sbannot]below left:$v_s$}] at (0,0) {};
\node[sbgoal,label={[sbannot]above right:$v_f$}] at (7,7) {};
% axes
\draw[->,black!60] (-0.6,-0.6) -- (8.2,-0.6) node[sbannot,anchor=north east,text=sbAgentA,yshift=-1pt] {progress of agent 1 along its own shortest path};
\draw[->,black!60] (-0.6,-0.6) -- (-0.6,8.2) node[sbannot,anchor=south east,rotate=90,text=sbAgentB,xshift=-2pt,yshift=8pt] {progress of agent 2};
% annotations
\node[sbannot,align=left,anchor=west,text=sbBlue] at (0.5,3.1) {both follow their\\ policies: a one-\\ dimensional path};
\node[sbannot,align=left,anchor=west,text=sbRed] at (3.45,2.35) {collision:\\ same cell at\\ the same time};
\node[sbannot,align=left,anchor=west,text=sbOrange] at (4.7,3.6) {detour inside the patch:\\ agent 2 waits one step,\\ then both continue};
\node[sbannot,align=left,anchor=west,text=sbBlue] at (4.2,6.3) {one-dimensional again};
\node[sbannot,align=center,anchor=north,text=sbGray] at (3.0,-1.25) {$|V|^2$ configurations in the lattice; \mstar visits the blue line, the patch, and nothing else};
\end{tikzpicture}

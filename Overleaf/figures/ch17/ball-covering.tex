% The ball-covering argument: a near-optimal path with clearance delta_n
% is covered by balls of radius q_n spaced along it; when every ball holds
% a tree vertex, consecutive vertices are within r_n of each other and the
% segments between them stay inside the clearance tube.
\begin{tikzpicture}[font=\footnotesize]
% obstacle and the tube of clearance delta_n around the path
\fill[sbobstacle] (2.6,0.2) rectangle (4.4,1.5);
\fill[sbBlue!12] plot[smooth,tension=0.8] coordinates {(0.3,0.3) (1.4,1.05) (2.4,1.85) (3.6,2.15) (4.9,2.05) (6.1,1.35) (7.2,0.6)}
  -- plot[smooth,tension=0.8] coordinates {(7.2,1.3) (6.2,2.05) (4.9,2.75) (3.6,2.85) (2.3,2.55) (1.2,1.75) (0.3,1.0)} -- cycle;
% the near-optimal path sigma_n
\draw[sbGreen!70!black,thick] plot[smooth,tension=0.8] coordinates {(0.3,0.65) (1.3,1.4) (2.35,2.2) (3.6,2.5) (4.9,2.4) (6.15,1.7) (7.2,0.95)};
% balls of radius q_n along the path
\foreach \x/\y in {0.3/0.65,0.95/1.15,1.6/1.65,2.25/2.13,2.95/2.4,3.65/2.5,4.35/2.47,5.05/2.35,5.7/2.0,6.3/1.6,6.8/1.25,7.2/0.95}{
  \draw[sbOrange,thin] (\x,\y) circle (0.36);}
% tree vertices in the balls and the tree path through them
\coordinate (p1) at (0.3,0.65); \coordinate (p2) at (1.05,1.35); \coordinate (p3) at (1.5,1.4);
\coordinate (p4) at (2.4,1.95); \coordinate (p5) at (3.1,2.55); \coordinate (p6) at (3.55,2.25);
\coordinate (p7) at (4.55,2.7); \coordinate (p8) at (4.95,2.1); \coordinate (p9) at (5.85,2.3);
\coordinate (p10) at (6.15,1.4); \coordinate (p11) at (6.95,1.45); \coordinate (p12) at (7.2,0.95);
\draw[sbPath] (p1)--(p2)--(p3)--(p4)--(p5)--(p6)--(p7)--(p8)--(p9)--(p10)--(p11)--(p12);
\foreach \i in {2,...,11}{\fill[sbBlue] (p\i) circle (1.5pt);}
\node[sbstart,minimum size=8pt] at (p1) {};
\node[sbgoal,minimum size=8pt] at (p12) {};
% annotations
\node[sbannot,below] at (0.3,0.4) {$x_{\mathrm{init}}$};
\node[sbannot,below] at (7.2,0.7) {$x_{\mathrm{goal}}$};
\node[sbannot,text=sbGreen!70!black] at (3.6,3.15) {near-optimal path $\sigma_n$ with clearance $\delta_n$};
\node[sbannot,text=sbOrange] at (1.1,2.55) {balls $B_{n,1},\dots,B_{n,M_n}$, radius $q_n$};
\draw[sbOrange,-{Stealth[length=4pt]}] (1.4,2.35) -- (1.55,1.95);
\node[sbannot,text=sbPath] at (6.2,0.35) {tree path through the balls};
\draw[sbPath,-{Stealth[length=4pt]}] (6.2,0.5) -- (6.1,1.3);
\node[sbannot] at (3.5,0.85) {obstacle};
\draw[sbGray,|-|,thin] (4.35,2.47) -- node[above,sbannot,yshift=-1pt]{$\le r_n$} (5.05,2.35);
\end{tikzpicture}

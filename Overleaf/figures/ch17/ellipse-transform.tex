% Sampling the informed set: a point of the unit ball is scaled by L,
% rotated by C and translated to the centre between the foci.  The six
% marked points are transformed by TikZ with exactly these operations
% (a = 1.5, b = 1, angle 30 degrees), so the panels are consistent.
\begin{tikzpicture}[font=\footnotesize,
  pt/.style={circle,fill=sbBlue,inner sep=1.1pt}]
% (1) the unit ball
\begin{scope}
\draw[sbGray,thin] (-1.3,0)--(1.3,0) (0,-1.3)--(0,1.3);
\draw[sbPurple,thick] (0,0) circle (1);
\foreach \x/\y in {0.5/0.2,-0.6/0.5,0.1/-0.8,-0.3/-0.3,0.8/-0.4,-0.1/0.9}{\node[pt] at (\x,\y) {};}
\node[sbannot] at (0,-1.75) {$\vect{x}_{\mathrm{ball}}$ uniform in the unit ball};
\end{scope}
\draw[sbflow] (1.55,0) -- node[above,sbannot]{$\mat{L}$} (2.55,0);
% (2) scaled: axes c_best/2 and sqrt(c_best^2-c_min^2)/2
\begin{scope}[shift={(4.2,0)}]
\draw[sbGray,thin] (-1.8,0)--(1.8,0) (0,-1.3)--(0,1.3);
\begin{scope}[xscale=1.5]
\draw[sbPurple,thick] (0,0) circle (1);
\foreach \x/\y in {0.5/0.2,-0.6/0.5,0.1/-0.8,-0.3/-0.3,0.8/-0.4,-0.1/0.9}{\node[pt] at (\x,\y) {};}
\end{scope}
\draw[sbOrange,thick,|-|] (0,-0.08) -- node[below,sbannot,text=sbOrange]{$c_{\mathrm{best}}/2$} (1.5,-0.08);
\draw[sbOrange,thick,|-|] (0.08,0) -- node[right,sbannot,text=sbOrange,align=left]{$\tfrac12\sqrt{c_{\mathrm{best}}^2-c_{\min}^2}$} (0.08,1);
\node[sbannot] at (0,-1.75) {$\mat{L}\vect{x}_{\mathrm{ball}}$: axis-aligned spheroid};
\end{scope}
\draw[sbflow] (6.35,0) -- node[above,sbannot,align=center]{$\mat{C}$, then\\$+\,\vect{x}_{\mathrm{centre}}$} (7.6,0);
% (3) rotated and translated: foci at the start and the goal
\begin{scope}[shift={(9.55,0.15)}]
\begin{scope}[rotate=30,xscale=1.5]
\draw[sbPurple,thick] (0,0) circle (1);
\foreach \x/\y in {0.5/0.2,-0.6/0.5,0.1/-0.8,-0.3/-0.3,0.8/-0.4,-0.1/0.9}{\node[pt] at (\x,\y) {};}
\draw[sbGray,thin] (-1.15,0)--(1.15,0);
\end{scope}
\node[sbstart,minimum size=7pt] at (-0.968,-0.559) {};
\node[sbgoal,minimum size=7pt] at (0.968,0.559) {};
\node[sbannot,below] at (-0.968,-0.68) {$x_{\mathrm{init}}$};
\node[sbannot,above] at (0.968,0.68) {$x_{\mathrm{goal}}$};
\fill[sbGray] (0,0) circle (1pt);
\node[sbannot,below right] at (0.02,-0.02) {$\vect{x}_{\mathrm{centre}}$};
\node[sbannot] at (0,-1.9) {$\mat{C}\mat{L}\vect{x}_{\mathrm{ball}}+\vect{x}_{\mathrm{centre}}$};
\end{scope}
\end{tikzpicture}

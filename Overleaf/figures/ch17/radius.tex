% The connection radius r_n = min(gamma (log n / n)^(1/d), eta) and the
% expected size of the Near set, n zeta_d r_n^d / mu(X_free), for the
% worked example (d = 2, mu = 76, gamma = 13.3, eta = 1) and for
% gamma = 20; plotted from the formulas by pgfplots.
\begin{tikzpicture}[font=\footnotesize]
\begin{groupplot}[group style={group size=2 by 1,horizontal sep=1.6cm},
  width=0.5\textwidth,height=4.8cm,grid=major,
  xmode=log,xmin=10,xmax=20000,xlabel={vertices $n$},
  legend style={font=\scriptsize,fill=white,fill opacity=0.9,draw=none},
  samples=200,domain=10:20000]
\nextgroupplot[ylabel={radius $r_n$},ymin=0,ymax=2.6,legend pos=north east]
\addplot[sbBlue,thick] {min(13.3*sqrt(ln(x)/x),1)};
\addplot[sbOrange,thick] {min(20*sqrt(ln(x)/x),1)};
\addplot[sbPurple,thick,dashed] {13.3*sqrt(ln(x)/x)};
\addplot[sbGray,thin] {1};
\legend{$\gamma=13.3$,$\gamma=20$,$\gamma=13.3$ uncapped,$\eta=1$}
\nextgroupplot[ylabel={expected $\abs{\Near}$},ymin=0,ymax=110,legend pos=north west]
\addplot[sbBlue,thick] {pi*min(13.3*sqrt(ln(x)/x),1)^2*x/76};
\addplot[sbOrange,thick] {pi*min(20*sqrt(ln(x)/x),1)^2*x/76};
\addplot[sbPurple,thick,dashed] {pi*13.3^2*ln(x)/76};
\legend{$\gamma=13.3$,$\gamma=20$,$\zeta_2\gamma^2\ln n/\mu$}
\end{groupplot}
\end{tikzpicture}

% Social pooling: the hidden states of the neighbours are summed per cell of
% a grid centred on the agent and fed to its LSTM together with its input.
% Grid coordinates are in units of 0.6 cm; the scope shifts are absolute.
\begin{tikzpicture}[font=\small,x=0.6cm,y=0.6cm]
  % --- left: the scene with the pooling grid around agent i
  \begin{scope}
    \draw[black!30,very thin] (-2,-2) grid (2,2);
    \draw[sbGray,thick] (-2,-2) rectangle (2,2);
    \node[sbagentA] (i) at (0,0) {$i$};
    \draw[sbvecA] (i) -- ++(0.0,1.0);
    \node[sbagentB] (j1) at (-1.4,1.4) {$j_1$};
    \draw[sbvecB] (j1) -- ++(0.7,0.4);
    \node[sbagentC] (j2) at (1.5,0.5) {$j_2$};
    \draw[sbvec,draw=sbAgentC] (j2) -- ++(-0.6,0.6);
    \node[sbagentD] (j3) at (1.3,-1.5) {$j_3$};
    \draw[sbvec,draw=sbAgentD] (j3) -- ++(0.0,0.8);
    \node[sbagentB,draw=sbGray,fill=sbGray!25] (j4) at (3.4,1.2) {$j_4$};
    \draw[sbvec,draw=sbGray] (j4) -- ++(-0.6,-0.5);
    \node[sbannot,anchor=north,align=center] at (0,-2.2) {$N_o \times N_o$ grid of side $2\rho$\\ around agent $i$};
    \node[sbannot,anchor=west,align=left] at (2.3,-1.1) {$j_4$: outside,\\ ignored};
    \node[sbannot,anchor=south west] at (-2.0,2.05) {$\vect{h}_{j_1}$};
    \node[sbannot,anchor=south west] at (1.1,0.95) {$\vect{h}_{j_2}$};
    \node[sbannot,anchor=north west] at (1.6,-1.6) {$\vect{h}_{j_3}$};
  \end{scope}
  % --- arrow
  \draw[sbflow] (3.0cm,0) -- (4.3cm,0) node[midway,above,sbannot] {pool};
  % --- middle: the pooling tensor H_i
  \begin{scope}[xshift=5.7cm]
    \draw[black!30,very thin] (-2,-2) grid (2,2);

    \node[font=\footnotesize,sbPurple] at (1.5,-1.5) {$\vect{h}_{j_3}$};

    \node[sbannot,anchor=north,align=center] at (0,-2.2) {$\mat{H}_i \in \R^{N_o \times N_o \times D}$\\ $\mat{H}_i[m,n,:] = \sum_{j \in \text{cell}(m,n)} \vect{h}_j$};
  \end{scope}
  % --- right: into the LSTM of agent i
  \draw[sbflow] (7.1cm,0) -- (8.1cm,0) node[midway,above,sbannot] {embed};

  \end{scope}

% Worked example of Chapter 8: three agents on a 3 x 7 grid, two priority orders.
% Cell (r,c) has its centre at (c+0.5, 2.5-r); row 0 is at the top.
\begin{tikzpicture}[x=0.72cm,y=0.72cm,font=\footnotesize,
  goalA/.style={circle,draw=sbAgentA,dashed,thick,inner sep=1pt,minimum size=13pt},
  goalB/.style={circle,draw=sbAgentB,dashed,thick,inner sep=1pt,minimum size=13pt},
  goalC/.style={circle,draw=sbAgentC,dashed,thick,inner sep=1pt,minimum size=13pt},
  pathA/.style={draw=sbAgentA,line width=1.3pt,-{Stealth[length=4pt]}},
  pathB/.style={draw=sbAgentB,line width=1.3pt,-{Stealth[length=4pt]}},
  pathC/.style={draw=sbAgentC,line width=1.3pt,-{Stealth[length=4pt]}},
  tA/.style={font=\tiny,text=sbAgentA},tB/.style={font=\tiny,text=sbAgentB},tC/.style={font=\tiny,text=sbAgentC}]
% ---------------- panel (a): order (1,2,3) ----------------
\begin{scope}
\drawgrid{7}{3}
\gridobstacle{1}{1}\gridobstacle{2}{1}\gridobstacle{4}{1}\gridobstacle{5}{1}
\foreach \c in {0,...,6} \node[sbannot,font=\tiny] at (\c+0.5,3.25) {$c{=}\c$};
\foreach \r in {0,...,2} \node[sbannot,font=\tiny,anchor=east] at (-0.1,2.5-\r) {$r{=}\r$};
\draw[pathA] (0.5,0.5) -- (6.2,0.5);
\draw[pathB] (0.5,2.5) -- (6.2,2.5);
\foreach \c in {1,...,6} {\node[tA] at (\c+0.5,0.2) {\c}; \node[tB] at (\c+0.5,2.8) {\c};}
\draw[pathC] (3.3,2.5) -- (3.3,0.8);
\draw[pathC,dashed] (3.55,0.8) -- (3.55,1.4);
\draw[pathC,dashed] (3.75,1.4) -- (3.75,0.8);
\node[tC] at (3.1,1.55) {1}; \node[tC] at (3.1,0.9) {2}; \node[tC] at (3.55,1.6) {3}; \node[tC] at (3.95,1.0) {4};
\node[goalA] at (6.5,0.5) {}; \node[goalB] at (6.5,2.5) {}; \node[goalC] at (3.5,0.5) {};
\node[sbagentA] at (0.5,0.5) {1}; \node[sbagentB] at (0.5,2.5) {2}; \node[sbagentC] at (3.5,2.5) {3};
\node[sbannot,anchor=north] at (3.5,-0.15) {(a) order $(1,2,3)$: costs $6+6+4=16$};
\end{scope}
% ---------------- panel (b): order (3,1,2) ----------------
\begin{scope}[xshift=8.4cm]
\drawgrid{7}{3}
\gridobstacle{1}{1}\gridobstacle{2}{1}\gridobstacle{4}{1}\gridobstacle{5}{1}
\foreach \c in {0,...,6} \node[sbannot,font=\tiny] at (\c+0.5,3.25) {$c{=}\c$};
% agent 3 first: straight down, parked at t=2
\draw[pathC] (3.5,2.5) -- (3.5,0.8);
\node[tC] at (3.75,1.85) {1}; \node[tC,anchor=west] at (3.65,0.55) {2: parked};
% agent 1: around through row 0
\draw[pathA] (0.62,0.5) -- (0.62,2.62) -- (6.62,2.62) -- (6.62,0.8);
\node[tA] at (0.85,1.5) {1}; \node[tA] at (0.85,2.2) {2}; \node[tA] at (3.5,2.9) {5}; \node[tA] at (6.85,2.2) {9}; \node[tA] at (6.85,1.0) {10};
% agent 2: waits at (0,3), dodges into (1,3) at t=5, returns, continues
\draw[pathB] (0.38,2.38) -- (3.25,2.38) -- (3.25,1.6);
\draw[pathB] (3.75,1.6) -- (3.75,2.38) -- (6.3,2.38);
\node[tB] at (2.9,2.15) {3,4}; \node[tB] at (3.05,1.4) {5}; \node[tB] at (3.98,1.4) {6}; \node[tB] at (5.5,2.15) {8}; \node[tB] at (6.5,2.12) {9};
\node[goalA] at (6.5,0.5) {}; \node[goalB] at (6.5,2.5) {}; \node[goalC] at (3.5,0.5) {};
\node[sbagentA] at (0.5,0.5) {1}; \node[sbagentB] at (0.5,2.5) {2}; \node[sbagentC] at (3.5,2.5) {3};
\node[sbannot,anchor=north] at (3.5,-0.15) {(b) order $(3,1,2)$: costs $10+9+2=21$};
\end{scope}
\end{tikzpicture}

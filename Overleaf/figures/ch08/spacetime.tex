% The reservation table as a space-time diagram: column c=3 of the worked example
% after agents 1 and 2 have been planned, and the path that agent 3 finds.
% Cell (i,3) at time t is the unit square with lower-left corner (t, 2-i).
\begin{tikzpicture}[x=0.9cm,y=0.9cm,font=\footnotesize]
\draw[black!40,very thin] (0,0) grid (7,3);
\foreach \t in {0,...,6} \node[sbannot] at (\t+0.5,3.3) {$t{=}\t$};
\node[sbannot,anchor=east] at (-0.15,2.5) {$(0,3)$};
\node[sbannot,anchor=east] at (-0.15,1.5) {$(1,3)$};
\node[sbannot,anchor=east] at (-0.15,0.5) {$(2,3)$};
% vertex reservations in this column: agent 2 at (0,3) at t=3, agent 1 at (2,3) at t=3
\fill[sbAgentB,opacity=0.35] (3,2) rectangle ++(1,1);
\fill[sbAgentA,opacity=0.35] (3,0) rectangle ++(1,1);
% agent 1 enters (2,3) from (2,2) during step 2->3 and leaves to (2,4) during 3->4
\draw[sbvecA] (2.3,-0.45) -- (3.25,0.3);
\draw[sbvecA] (3.75,0.3) -- (4.7,-0.45);
\node[sbannot,text=sbAgentA,anchor=east] at (2.35,-0.5) {\scriptsize from $(2,2)$};
\node[sbannot,text=sbAgentA,anchor=west] at (4.65,-0.5) {\scriptsize to $(2,4)$};
% agent 3 stays at its goal from t=4 on
\fill[sbAgentC,opacity=0.18] (4,0) rectangle (7,1);
% path of agent 3
\draw[draw=sbAgentC,line width=1.4pt,-{Stealth[length=4pt]}] (0.5,2.5) -- (1.5,1.5) -- (2.5,0.5) -- (3.5,1.5) -- (4.5,0.5) -- (6.8,0.5);
\foreach \t/\y in {0/2.5,1/1.5,2/0.5,3/1.5,4/0.5} \node[sbagentC] at (\t+0.5,\y) {3};
\node[sbagentB] at (3.5,2.5) {2};
\node[sbagentA] at (3.5,0.5) {1};
% forbidden successors of (2,3) at t=2 and of (1,3) at t=2
\draw[sbRed,dashed,thick,-{Stealth[length=4pt]}] (2.75,0.3) -- (3.3,0.3);
\draw[sbRed,dashed,thick,-{Stealth[length=4pt]}] (2.7,0.25) -- (3.2,-0.4);
\draw[sbRed,dashed,thick,-{Stealth[length=4pt]}] (2.75,1.7) -- (3.3,2.3);
\node[sbannot,text=sbRed,anchor=west] at (3.15,-0.55) {\scriptsize move to $(2,2)$: swap with agent 1};
\node[sbannot,text=sbRed,anchor=west,align=left] at (7.3,0.75) {\scriptsize wait at $(2,3)$: vertex\\[-2pt] \scriptsize reserved by agent 1};
\node[sbannot,text=sbRed,anchor=west,align=left] at (7.3,2.4) {\scriptsize move up: $(0,3)$ reserved\\[-2pt] \scriptsize by agent 2 at $t=3$};
\node[sbannot,anchor=west,align=left] at (7.3,1.55) {shaded: entries of $R$\\[-1pt] in column $c=3$; green:\\[-1pt] agent 3 parked, $t_\gamma=3$};
\end{tikzpicture}

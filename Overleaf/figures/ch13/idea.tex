% The reciprocal dance: two agents that both avoid with plain velocity
% obstacles alternate between swerving and returning; with reciprocity each
% takes half of the swerve and keeps it.
\begin{tikzpicture}[font=\footnotesize]
\foreach \k/\ya/\yb/\lab in {%
  0/0.32/-0.32/{$t$: $\vel_A \in \VO_{A|B}$ and $\vel_B \in \VO_{B|A}$; both swerve by the full amount that clears the cone},
  1/0/0/{$t + \dt$: the other agent's new velocity has moved the cone away; $\vel^{\mathrm{pref}}$ is allowed again and both return to it},
  2/0.32/-0.32/{$t + 2\dt$: head-on again; both swerve again, and so on}}{
  \begin{scope}[yshift=-\k*1.45cm]
    \draw[black!25,dashed] (-0.2,0) -- (4.0,0);
    \node[sbagentA] (a) at (0.3,0) {A};
    \node[sbagentB] (b) at (3.5,0) {B};
    \draw[sbvecA] (a.center) -- ++(1.1,\ya);
    \draw[sbvecB] (b.center) -- ++(-1.1,\yb);
    \node[sbannot,anchor=west,text width=8.4cm] at (4.3,0) {\lab};
  \end{scope}
}
\begin{scope}[yshift=-4.6cm]
  \draw[black!25,dashed] (-0.2,0) -- (4.0,0);
  \node[sbagentA] (a) at (0.3,0) {A};
  \node[sbagentB] (b) at (3.5,0) {B};
  \draw[sbvecA] (a.center) -- ++(1.1,0.16);
  \draw[sbvecB] (b.center) -- ++(-1.1,-0.16);
  \node[sbannot,anchor=west,text width=8.4cm] at (4.3,0) {reciprocal (RVO, \orca): each agent takes half of the swerve; the relative velocity still clears the cone, and at $t + \dt$ nothing has changed, so both keep their velocities};
\end{scope}
\begin{scope}[yshift=-6.4cm]
  \draw[sbpathalt] (0,0.15) -- (0.5,0.33) -- (1.0,0.33) -- (1.5,0.51) -- (2.0,0.51) -- (2.5,0.69) -- (3.0,0.69) -- (3.5,0.87);
  \node[sbannot,anchor=west] at (4.3,0.55) {VO: a zig-zag, the velocity changes at every step};
  \draw[sbpath] (0,-0.45) -- (3.5,-0.09);
  \node[sbannot,anchor=west] at (4.3,-0.3) {\orca: one smooth swerve};
\end{scope}
\end{tikzpicture}

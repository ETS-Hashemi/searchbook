% Velocity space of agent A in a head-on encounter, v_A = (1,0), v_B = (-1,0),
% half-angle 20 degrees.  The velocity obstacle VO_{A|B}(v_B) has its apex at
% v_B; the reciprocal velocity obstacle RVO_{A|B}(v_B, v_A) is the same cone
% with the apex moved to (v_A + v_B)/2.
\begin{tikzpicture}[scale=1.7,font=\footnotesize]
\clip (-1.75,-1.25) rectangle (3.55,1.6);
\fill[sbOrange!18] (-1,0) -- ++(20:4.6) -- ++(0,-3.146) -- cycle;
\draw[sbOrange,thick] (-1,0) ++(20:4.6) -- (-1,0) -- ++(-20:4.6);
\fill[sbBlue!15,opacity=0.85] (0,0) -- ++(20:3.6) -- ++(0,-2.463) -- cycle;
\draw[sbBlue,thick,dashed] (0,0) ++(20:3.6) -- (0,0) -- ++(-20:3.6);
\draw[black!40,->] (-1.65,0) -- (3.4,0) node[below left] {$v_x$};
\draw[black!40,->] (0,-1.15) -- (0,1.5) node[left] {$v_y$};
\draw[sbvecB] (0,0) -- (-1,0);
\draw[sbvecA] (0,0) -- (1,0);
\fill[sbOrange] (-1,0) circle (1.3pt) node[below,align=center] {$\vel_B$\\apex of $\VO$};
\fill[sbBlue] (0,0) circle (1.3pt) node[below right=1pt,align=left] {$\tfrac12(\vel_A+\vel_B)$\\apex of $\RVO$};
\coordinate (vA) at (1,0);
\coordinate (vVO) at (0.766,0.643);
\coordinate (vRVO) at (0.883,0.321);
\fill[sbBlue] (vA) circle (1.3pt) node[below right=1pt] {$\vel_A = \vel_A^{\mathrm{pref}}$};
\draw[black!50,dotted,thick] (vA) -- (vVO);
\fill[sbOrange] (vVO) circle (1.5pt) node[above left=-1pt] {$\vel' \notin \VO$};
\fill[sbBlue] (vRVO) circle (1.5pt) node[right=2pt] {$\vel_A^{\mathrm{new}} = \tfrac12(\vel_A + \vel')\notin\RVO$};
\node[sbannot,text=sbOrange,anchor=west] at (2.3,1.35) {$\VO_{A|B}(\vel_B)$};
\node[sbannot,text=sbBlue,anchor=west] at (2.6,0.75) {$\RVO_{A|B}(\vel_B,\vel_A)$};
\node[sbannot,anchor=west] at (1.3,-0.75) {the RVO exit is half as far from $\vel_A$ as the VO exit};
\end{tikzpicture}

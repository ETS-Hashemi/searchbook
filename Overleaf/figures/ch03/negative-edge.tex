% The negative-edge counterexample (NEGATIVE_EXAMPLE in code/ch03_dijkstra.py).
% Dijkstra settles A with g = 1 before it discovers the cheaper route S-B-A.
\begin{tikzpicture}[scale=1.1,every node/.style={font=\small}]
  \node[sbnode,draw=sbStart,fill=sbStart!20] (S) at (0,0) {S};
  \node[sbnode] (A) at (2.6,1.2) {A};
  \node[sbnode] (B) at (2.6,-1.2) {B};
  \node[sbnode,draw=sbGoal,fill=sbGoal!20] (T) at (5.2,0) {T};
  % route overlays (drawn first, under the edges)
  \draw[sbpath,opacity=0.35,line width=5pt] (S) to[bend left=12] (A) to[bend left=12] (T);
  \draw[sbpathalt,opacity=0.55,line width=4pt] (S) to[bend right=12] (B) to[bend right=25] (A) to[bend left=12] (T);
  \draw[sbedge,->] (S) -- node[fill=white,inner sep=1.5pt,font=\footnotesize] {1} (A);
  \draw[sbedge,->] (S) -- node[fill=white,inner sep=1.5pt,font=\footnotesize] {2} (B);
  \draw[sbedge,->] (A) -- node[fill=white,inner sep=1.5pt,font=\footnotesize] {1} (T);
  \draw[sbedge,->,draw=sbRed,line width=1.2pt] (B) -- node[fill=white,inner sep=1.5pt,font=\footnotesize,text=sbRed] {$-2$} (A);
  \node[sbannot,above=3pt of A,align=center] {settled second with $\gcost=1$\\ but $\delta(S,A)=0$};
  \node[sbannot,below=3pt of B,align=center] {settled third with $\gcost=2$;\\ B$\to$A finds $0<1$, but A is closed};
  \node[sbannot,below=3pt of T,align=center] {settled fourth with $\gcost=2$\\ true cost $1$};
  \node[sbannot,below=3pt of S] {$\gcost=0$};
  \draw[sbpath,line width=2.5pt,opacity=0.5] (7.2,1.0) -- (7.9,1.0);
  \node[sbannot,anchor=west] at (8.0,1.0) {returned: S--A--T, cost 2};
  \draw[sbpathalt,line width=2.5pt,opacity=0.7] (7.2,0.4) -- (7.9,0.4);
  \node[sbannot,anchor=west] at (8.0,0.4) {optimal: S--B--A--T, cost 1};
\end{tikzpicture}

% the base graph G: a path a - b - c

  \node[sbnode] (A) at (0,0) {$a$};
  \node[sbnode] (B) at (0,1.6) {$b$};
  \node[sbnode] (C) at (0,3.2) {$c$};

    \node[sbnode] (A\t) at (2.6*\t,0) {$a$};

    \node[sbnode] (C\t) at (2.6*\t,3.2) {$c$};
    \node[sbannot] at (2.6*\t,4.0) {$t=\t$};
  }
  \foreach \t/\s in {0/1,1/2}{
    % wait edges (grey)
    \draw[sbedge,draw=black!35,->] (A\t) -- (A\s);
    \draw[sbedge,draw=black!35,->] (B\t) -- (B\s);
    \draw[sbedge,draw=black!35,->] (C\t) -- (C\s);
    % move edges
    \draw[sbedge,->] (A\t) -- (B\s);
    \draw[sbedge,->] (B\t) -- (A\s);
    \draw[sbedge,->] (B\t) -- (C\s);
    \draw[sbedge,->] (C\t) -- (B\s);
  }
  % a time-indexed path: wait at a, then move to b
  \draw[sbedgeopt,->] (A0) -- (A1);
  \draw[sbedgeopt,->] (A1) -- (B2);

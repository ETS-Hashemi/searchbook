% subdivision collision check (hand-drawn).
\begin{tikzpicture}[scale=1.0]
% obstacle and its inflated copy (margin r + delta/2)
\fill[sbobstacle] (4.6,0.4) rectangle (6.2,2.4);
\draw[sbOrange,dashed,thin] (4.35,0.15) rectangle (6.45,2.65);
\node[sbannot,anchor=south west] at (4.35,2.72) {$\Xobs$, inflated by $r+\delta/2$ (dashed)};
% the tree
\draw[sbedge] (0.3,0.4)--(1.0,1.1)--(1.9,1.4)--(2.6,2.1);
\draw[sbedge] (1.0,1.1)--(1.4,0.3);
\draw[sbedge] (1.9,1.4)--(2.9,1.0);
\foreach \p in {(1.0,1.1),(1.9,1.4),(2.6,2.1),(1.4,0.3),(2.9,1.0)} {\fill[sbBlue] \p circle (1.6pt);}
\node[sbstart] at (0.3,0.4) {};
\node[sbannot,anchor=east] at (0.12,0.4) {$x_{\mathrm{init}}$};
% first extension: accepted
\coordinate (near) at (2.9,1.0);
\coordinate (rand) at (6.9,2.6);
\coordinate (new) at ($(near)!1.2cm!(rand)$);
\draw[sbGray,dashed,thin] (near) -- (rand);
\fill[sbRed] (rand) circle (1.8pt);
\node[sbannot,anchor=west] at ($(rand)+(0.1,0)$) {$x_{\mathrm{rand}}$};
\draw[sbOrange,line width=1.4pt] (near) -- (new);
\foreach \t in {0,0.25,0.5,0.75,1} {\fill[white,draw=sbOrange,thin] ($(near)!\t!(new)$) circle (1.2pt);}
\fill[sbOrange] (new) circle (1.8pt);
\node[sbannot,anchor=north] at ($(near)+(0,-0.12)$) {$x_{\mathrm{near}}$};
\node[sbannot,anchor=north west] at ($(new)+(0.02,-0.1)$) {$x_{\mathrm{new}}$};
\draw[decorate,decoration={brace,amplitude=4pt,raise=6pt}] (near) -- (new) node[midway,above=9pt,sbannot] {$\eta$};
\draw[sbannot,<->,thin] ($(near)!0.5!(new)+(0,-0.4)$) -- ($(near)!0.75!(new)+(0,-0.4)$) node[midway,below,sbannot] {$\delta$};
% second extension: rejected at its second test point
\coordinate (rand2) at (6.9,1.2);
\coordinate (new2) at ($(new)!1.2cm!(rand2)$);
\draw[sbGray,dashed,thin] (new) -- (rand2);
\fill[sbRed] (rand2) circle (1.8pt);
\node[sbannot,anchor=west] at ($(rand2)+(0.1,0)$) {$x'_{\mathrm{rand}}$};
\draw[sbRed,line width=1.0pt,dotted] (new) -- (new2);
\fill[white,draw=sbRed,thin] ($(new)!0.25!(new2)$) circle (1.2pt);
\node[sbRed,font=\small] at ($(new)!0.5!(new2)$) {$\times$};
\node[sbannot,anchor=north,text=sbRed] at (5.4,-0.05) {second step rejected};
\end{tikzpicture}

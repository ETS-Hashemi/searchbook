% Concept figure: the communication graph of a swarm (left) and the
% consensus rule "move toward your neighbors" (right).
\begin{tikzpicture}[scale=1.0]
% ---- left: radius graph -------------------------------------------
\begin{scope}
  \coordinate (A) at (0,0);
  \coordinate (B) at (1.3,0.9);
  \coordinate (C) at (2.6,0.2);
  \coordinate (D) at (3.9,1.1);
  \coordinate (E) at (3.2,-1.2);
  \coordinate (F) at (5.4,0.4);
  \draw[sbGray,dashed,thin] (C) circle (1.7);
  \node[sbannot] at (2.6,2.15) {$R_{\mathrm{comm}}$};
  \draw[sbGray,thin,-{Stealth[length=4pt]}] (C) -- ++(-1.7,0);
  \draw[sbedge] (A) -- (B);
  \draw[sbedge] (B) -- (C);
  \draw[sbedge] (C) -- (D);
  \draw[sbedge] (C) -- (E);
  \draw[sbedge] (D) -- (F);
  \node[sbagentA] at (A) {1};
  \node[sbagentA] at (B) {2};
  \node[sbagentB] at (C) {3};
  \node[sbagentA] at (D) {4};
  \node[sbagentA] at (E) {5};
  \node[sbagentA] at (F) {6};
  \node[sbannot,align=center] at (2.7,-2.0) {$N_3=\{2,4,5\}$, $\deg(3)=3$\\ $(3,6)\notin E$: too far apart};
\end{scope}
% ---- right: consensus rule ----------------------------------------
\begin{scope}[xshift=8.6cm]
  \coordinate (I) at (0,0);
  \coordinate (J1) at (1.5,0.5);
  \coordinate (J2) at (0.8,1.4);
  \coordinate (J3) at (-0.6,-1.2);
  \draw[sbGray,dashed,thin] (I) circle (1.7);
  \draw[sbedge,black!35] (I) -- (J1);
  \draw[sbedge,black!35] (I) -- (J2);
  \draw[sbedge,black!35] (I) -- (J3);
  \draw[sbvec,draw=sbGray] (I) -- ($(I)!0.55!(J1)$);
  \draw[sbvec,draw=sbGray] (I) -- ($(I)!0.55!(J2)$);
  \draw[sbvec,draw=sbGray] (I) -- ($(I)!0.55!(J3)$);
  \draw[sbvec,draw=sbOrange,line width=1.6pt] (I) -- (1.02,0.42);
  \node[sbagentB] at (I) {$i$};
  \node[sbagentA] at (J1) {$j$};
  \node[sbagentA] at (J2) {$j'$};
  \node[sbagentA] at (J3) {$j''$};
  \node[sbannot,text=sbOrange,anchor=west] at (1.1,0.05) {$\dot{\vect{x}}_i=\sum_{j\in N_i}(\vect{x}_j-\vect{x}_i)$};
  \node[sbannot,align=center] at (0.4,-2.0) {each drone moves toward the\\ average of its neighbors};
\end{scope}
\end{tikzpicture}

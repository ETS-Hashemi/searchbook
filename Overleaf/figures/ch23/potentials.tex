% Pairwise potentials as a function of the edge length: the quadratic
% displacement potential, the safety repulsion and the range-keeping
% barrier that becomes infinite at R_comm.
\begin{tikzpicture}
\begin{axis}[width=0.8\textwidth,height=5.6cm,font=\footnotesize,grid=major,
  xmin=0,xmax=3.4,ymin=0,ymax=2.6,xlabel={edge length $\ell=\norm{\pos_i-\pos_j}$ [m]},ylabel={potential},
  legend style={at={(0.03,0.97)},anchor=north west,font=\scriptsize},legend cell align=left,
  xtick={0,0.8,1.2,2,2.95,3.2},xticklabels={0,$d_{\mathrm{safe}}$,$\rho_0$,$d_{ij}$,$\ell_{\mathrm{act}}$,$R_{\mathrm{comm}}$}]
\addplot[sbBlue,thick,dashed,domain=0:3.4,samples=60] {0.5*(x-2)^2};
\addplot[sbGreen,thick,domain=0.55:1.2,samples=80] {0.5*(1/x-1/1.2)^2};
\addplot[sbOrange,thick,domain=2.95:3.135,samples=80] {(x-2.95)^2/(3.2-x)};
\addplot[sbOrange,thick,forget plot,domain=0:2.95,samples=2] {0};
\addplot[sbRed,dashed,thick,forget plot] coordinates {(3.2,0) (3.2,2.6)};
\addplot[sbRed,dotted,thick,forget plot] coordinates {(0.8,0) (0.8,2.6)};
\legend{displacement term $\tfrac12(\ell-d_{ij})^2$,safety repulsion $\tfrac12(1/\ell-1/\rho_0)^2$,range barrier $(\ell-\ell_{\mathrm{act}})^2/(R_{\mathrm{comm}}-\ell)$}
\node[sbannot,text=sbRed,anchor=east] at (axis cs:3.15,2.3) {$\to\infty$};
\end{axis}
\end{tikzpicture}

% The one-dimensional update: prior N(2, 1), likelihood of z = 3 with
% variance 0.25, posterior N(2.8, 0.2).  Densities computed by pgfplots.
\begin{tikzpicture}
\begin{axis}[
  width=0.66\textwidth,height=4.4cm,
  xlabel={position $x$ [m]},ylabel={density},
  xmin=-1,xmax=5.5,ymin=0,ymax=1.1,
  grid=major,font=\footnotesize,samples=161,domain=-1:5.5,
  legend pos=north west,legend cell align=left,legend style={font=\scriptsize},
]
\addplot[sbGray,thick,dashed] {exp(-(x-2)^2/2)/sqrt(2*pi)};
\addplot[sbRed,thick,dotted] {exp(-(x-3)^2/(2*0.25))/sqrt(2*pi*0.25)};
\addplot[sbBlue,line width=1.4pt] {exp(-(x-2.8)^2/(2*0.2))/sqrt(2*pi*0.2)};
\legend{{prediction $\Normal(2,\,1)$},{likelihood of $z=3$: $\Normal(3,\,0.25)$},{posterior $\Normal(2.8,\,0.2)$}}
\draw[sbGray,thin] (axis cs:2,0) -- (axis cs:2,0.40);
\draw[sbRed,thin] (axis cs:3,0) -- (axis cs:3,0.80);
\draw[sbBlue,thin] (axis cs:2.8,0) -- (axis cs:2.8,0.89);
\draw[sbvec,draw=sbBlue] (axis cs:2,1.0) -- (axis cs:2.8,1.0);
\node[sbannot,text=sbBlue,anchor=south] at (axis cs:2.4,1.0) {$K\,y=0.8\cdot 1$};
\node[sbannot,anchor=north] at (axis cs:2,0.02) {$\hat x^-$};
\node[sbannot,text=sbRed,anchor=north] at (axis cs:3.05,0.02) {$z$};
\node[sbannot,text=sbBlue,anchor=north] at (axis cs:2.75,0.02) {$\hat x$};
\end{axis}
\end{tikzpicture}

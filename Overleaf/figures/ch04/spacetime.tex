% Space-time A* mini example: a vertex constraint on cell (1,1) at t = 1
% forces the agent to wait one step.  Left: the grid.  Right: the states
% (cell, t) of the middle row for t = 0..3 with wait (horizontal) and move
% (diagonal) edges.
\begin{tikzpicture}[font=\small,
  st/.style={circle,draw=black!60,fill=white,inner sep=1pt,minimum size=8pt},
  stx/.style={st,fill=sbClosed}]
% left: the grid
\begin{scope}[scale=0.8]
  \drawgrid{3}{3}
  \fill[pattern=north east lines,pattern color=sbRed!80] (1,1) rectangle ++(1,1);
  \node[font=\scriptsize,text=sbRed,fill=white,inner sep=1pt] at (1.5,1.5) {$t{=}1$};
  \node[sbstart,label={[sbannot]above:$s$}] at (0.5,1.5) {};
  \node[sbgoal,label={[sbannot]above:$\gamma$}] at (2.5,1.5) {};
  \foreach \x in {0,1,2} \node[sbannot] at (\x+0.5,-0.4) {$(\x,1)$};
  \node[sbannot,text=black,align=center] at (1.5,-1.3) {the hatched cell is\\ forbidden at $t = 1$};
\end{scope}
% right: the middle row of the time-expanded graph
\begin{scope}[xshift=4.6cm]
  \foreach \t in {0,1,2,3} \node[sbannot] at (2*\t,3.4) {$t = \t$};
  \node[sbannot,anchor=east] at (-0.5,0) {$(0,1)$};
  \node[sbannot,anchor=east] at (-0.5,1.3) {$(1,1)$};
  \node[sbannot,anchor=east] at (-0.5,2.6) {$(2,1)$};
  \foreach \t in {0,1,2} {
    \pgfmathtruncatemacro{\s}{\t+1}
    \draw[black!30,->] (2*\t,0) -- (2*\s,0);
    \draw[black!30,->] (2*\t,1.3) -- (2*\s,1.3);
    \draw[black!30,->] (2*\t,2.6) -- (2*\s,2.6);
    \draw[black!30,->] (2*\t,0) -- (2*\s,1.3);
    \draw[black!30,->] (2*\t,1.3) -- (2*\s,0);
    \draw[black!30,->] (2*\t,1.3) -- (2*\s,2.6);
    \draw[black!30,->] (2*\t,2.6) -- (2*\s,1.3);
  }
  % the unconstrained path (dashed) and the constrained path (solid)
  \draw[draw=sbOrange,line width=1.6pt,dashed,->] (0,0) -- (2,1.3) -- (4,2.6);
  \draw[sbedgeopt,->] (0,0) -- (2,0);
  \draw[sbedgeopt,->] (2,0) -- (4,1.3) -- (6,2.6);
  % nodes: expanded states grey, the forbidden state crossed out in red
  \foreach \t in {0,1,2,3} { \node[st] at (2*\t,0) {}; \node[st] at (2*\t,1.3) {}; \node[st] at (2*\t,2.6) {}; }
  \node[stx] at (0,0) {}; \node[stx] at (2,0) {}; \node[stx] at (4,1.3) {}; \node[stx] at (6,2.6) {};
  \node[st,draw=sbRed,line width=1pt] (blk) at (2,1.3) {};
  \draw[sbRed,line width=1pt] (blk.north west) -- (blk.south east) (blk.north east) -- (blk.south west);
  % f = t + h(v) of the expanded states
  \node[sbannot,text=sbBlue,below=2pt] at (0,0) {$\fcost = 2$};
  \node[sbannot,text=sbBlue,below=2pt] at (2,0) {$\fcost = 3$};
  \node[sbannot,text=sbBlue,below right=1pt and -2pt] at (4,1.3) {$\fcost = 3$};
  \node[sbannot,text=sbBlue,above=2pt] at (6,2.6) {$\fcost = 3$};
  \node[sbannot,text=sbBlue] at (1,-0.55) {wait};
  \node[sbannot,text=sbOrange,anchor=west] at (2.15,2.0) {forbidden at $t=1$};
\end{scope}
\end{tikzpicture}

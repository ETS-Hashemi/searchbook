% Notation figure of Chapter 19: the coordinated-turn model and the
% range-bearing sensor.  Hand-drawn (no data).
\begin{tikzpicture}[x=1cm,y=1cm,font=\footnotesize]
  % axes of the sensor frame
  \draw[->,black!60] (-0.6,0) -- (6.2,0) node[below] {$x$};
  \draw[->,black!60] (0,-0.6) -- (0,4.2) node[left] {$y$};
  % turn circle: centre (2.42,2.22), radius 1.6; intruder at angle 10 deg
  \coordinate (c) at (2.42,2.22);
  \coordinate (p) at (4.0,2.5);
  \draw[sbGray,dashed] (c) ++(-70:1.6) arc (-70:10:1.6);
  \draw[sbGray,dashed,-{Stealth[length=5pt]}] (c) ++(10:1.6) arc (10:150:1.6);
  \draw[black!40,very thin] (c) -- (p);
  \fill[black!50] (c) circle (1pt);
  \node[sbannot,anchor=north east,inner sep=1pt] at (c) {centre};
  \node[sbannot,anchor=south west,inner sep=1pt] at ($(c)!0.5!(p)$) {$v/\abs{\omega}$};
  % turn-rate arrow near the arc
  \draw[sbOrange,thick,-{Stealth[length=5pt]}] (c) ++(60:2.05) arc (60:110:2.05);
  \node[sbannot,text=sbOrange!80!black] at ($(c)+(85:2.4)$) {$\omega > 0$ (left turn)};
  % range line and bearing angle
  \draw[sbBlue,thick] (0,0) -- node[above left,sbannot,text=sbBlue] {$r$} (p);
  \draw[sbBlue] (1.1,0) arc (0:32:1.1);
  \node[sbannot,text=sbBlue] at (1.5,0.35) {$\varphi$};
  % heading arrow and heading angle at the intruder
  \draw[black!40,dotted] (p) -- ++(1.4,0);
  \draw[sbvec,draw=sbRed] (p) -- ++(100:1.3) node[right,sbannot,text=sbRed,pos=0.9] {$v$};
  \draw[sbRed] (p) ++(0.7,0) arc (0:100:0.7);
  \node[sbannot,text=sbRed] at ($(p)+(50:0.95)$) {$\psi$};
  % sensor and intruder markers
  \node[sbagentA] at (0,0) {};
  \node[sbannot,anchor=north east] at (-0.1,-0.15) {sensor $\vect{s}$};
  \node[sbintruder] at (p) {};
  \node[sbannot,anchor=north west] at ($(p)+(0.12,-0.12)$) {intruder $(x, y)$};
  % legend on the right
  \node[sbannot,text=black,align=left,anchor=north west] at (6.5,4.0)
    {state $\state = (x, y, v, \psi, \omega)\T$\\[2pt]
     $\dot{x} = v\cos\psi$, $\dot{y} = v\sin\psi$, $\dot{\psi} = \omega$\\[2pt]
     path: arc of radius $v/\abs{\omega}$\\[6pt]
     measurement $\meas = (r, \varphi)\T$\\[2pt]
     $r = \norm{(x, y) - \vect{s}}$\\[2pt]
     $\varphi = \operatorname{atan2}(y - s_y,\, x - s_x)$};
\end{tikzpicture}

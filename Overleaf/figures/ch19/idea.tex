% Concept figure of Chapter 19: three ways of pushing a Gaussian through a
% nonlinear function.  Hand-drawn schematic (no data).  The input belief is an
% ellipse in (u, w) coordinates; the map is u -> angle 90 + 28u degrees,
% w -> radius 3 + 0.7w, so the output is a banana.
\begin{tikzpicture}[x=1cm,y=1cm,font=\footnotesize]
  % column headings
  \node[sbannot,text=black,align=center] at (0,1.75) {input belief\\ $\Normal(\hat{\state}, \mat{P})$};
  \node[sbannot,text=black,align=center] at (7.0,1.75) {output: true distribution of $f(\state)$\\ and each filter's approximation};
  \foreach \row/\yy/\lab in {a/0/{(a) EKF: linearise},b/-1.9/{(b) UKF: sigma points},c/-3.8/{(c) PF: samples}}{
    \begin{scope}[yshift=\yy cm]
      % input ellipse
      \draw[sbGray,thick] (0,0) ellipse (1.2 and 0.4);
      % arrow with function label
      \draw[sbflow] (1.5,0) -- node[above,sbannot,text=black] {$f$} (3.6,0);
      % banana: arc of radius 3 about (7,-2.6), angles 45..135, thickness 0.7
      \fill[sbRed!12] ($(7,-2.6)+(45:3.35)$) arc (45:135:3.35) -- ($(7,-2.6)+(135:2.65)$) arc (135:45:2.65) -- cycle;
      \draw[sbRed!70] ($(7,-2.6)+(45:3.35)$) arc (45:135:3.35) -- ($(7,-2.6)+(135:2.65)$) arc (135:45:2.65) -- cycle;
      \node[sbannot,text=black,anchor=east] at (-1.45,0) {\lab};
    \end{scope}
  }
  % ---- row (a): EKF ----
  \begin{scope}[yshift=0cm]
    \fill[sbBlue] (0,0) circle (1.6pt);
    \draw[sbBlue,thick] (7,0.4) ellipse (1.55 and 0.33);
    \fill[sbBlue] (7,0.4) circle (1.6pt);
    \draw[sbBlue,dashed] (5.45,0.4) -- (8.55,0.4);
    \node[sbannot,text=sbBlue,anchor=south] at (7,0.78) {tangent image of the mean};
    % true mean of the banana = centroid of the drawn annular sector
    \draw[sbRed!80!black,thick] (6.88,-0.01) -- (7.12,0.23) (6.88,0.23) -- (7.12,-0.01);
    \draw[sbRed!80!black,dotted] (7,-0.01) -- (7,-0.26);
    \node[sbannot,text=sbRed!80!black,anchor=north] at (7,-0.26) {true mean};
    % the gap between the two: the EKF's outward bias
    \draw[sbBlue,dotted] (8.55,0.40) -- (9.50,0.40);
    \draw[sbRed!80!black,dotted] (7.12,0.11) -- (9.50,0.11);
    \draw[<->,>=latex,sbRed!80!black] (9.38,0.40) -- (9.38,0.11);
    \node[sbannot,text=sbRed!80!black,anchor=west] at (9.54,0.255) {bias};
  \end{scope}
  % ---- row (b): UKF ----
  \begin{scope}[yshift=-1.9cm]
    \foreach \p in {(0,0),(1.2,0),(-1.2,0),(0,0.4),(0,-0.4)}{\fill[sbOrange] \p circle (1.8pt);}
    % images of the sigma points: (u,w) -> (7 + R cos(90+28u), -2.6 + R sin(90+28u)), R = 3 + 0.7 w
    \foreach \u/\w in {0/0,1.2/0,-1.2/0,0/0.4,0/-0.4}{
      \fill[sbOrange] ({7+(3+0.7*\w)*cos(90+28*\u)},{-2.6+(3+0.7*\w)*sin(90+28*\u)}) circle (1.8pt);
    }
    \draw[sbRed!80!black,dotted] (5.05,0.11) -- (9.10,0.11);
    \draw[sbOrange,thick] (7,0.20) ellipse (1.75 and 0.42);
    \fill[sbOrange] (7,0.20) circle (1.6pt);
    \node[sbannot,text=sbOrange!80!black,anchor=south] at (7,0.78) {ellipse fitted to the images};
  \end{scope}
  % ---- row (c): PF ----
  \begin{scope}[yshift=-3.8cm]
    \pgfmathsetseed{1919}
    \foreach \i in {1,...,70}{
      \pgfmathsetmacro{\u}{0.55*(rand+rand+rand)}
      \pgfmathsetmacro{\w}{0.18*(rand+rand+rand)}
      \fill[sbPurple] (\u,\w) circle (0.8pt);
      \fill[sbPurple] ({7+(3+0.7*\w)*cos(90+28*\u)},{-2.6+(3+0.7*\w)*sin(90+28*\u)}) circle (0.8pt);
    }
    \node[sbannot,text=sbPurple,anchor=south] at (7,0.78) {the cloud is the belief};
  \end{scope}
\end{tikzpicture}

% Concept figure of Chapter 19: three ways of pushing a Gaussian through a
% nonlinear function.  Hand-drawn schematic (no data).  The input belief is an
% ellipse in (u, w) coordinates; the map is u -> angle 90 + 20u degrees,
% w -> radius 3 + 0.7w, so the output is a banana.
\begin{tikzpicture}[x=1cm,y=1cm,font=\footnotesize]
  % column headings
  \node[sbannot,text=black,align=center] at (0,1.0) {input belief\\ $\Normal(\hat{\state}, \mat{P})$};
  \node[sbannot,text=black,align=center] at (7.0,1.0) {output: true distribution of $f(\state)$\\ and each filter's approximation};
  \foreach \row/\yy/\lab in {a/0/{(a) EKF: linearise},b/-1.9/{(b) UKF: sigma points},c/-3.8/{(c) PF: samples}}{
    \begin{scope}[yshift=\yy cm]
      % input ellipse
      \draw[sbGray,thick] (0,0) ellipse (1.2 and 0.4);
      % arrow with function label
      \draw[sbflow] (1.5,0) -- node[above,sbannot,text=black] {$f$} (3.6,0);
      % banana: arc of radius 3 about (7,-2.6), angles 60..120, thickness 0.7
      \fill[sbRed!12] ($(7,-2.6)+(60:3.35)$) arc (60:120:3.35) -- ($(7,-2.6)+(120:2.65)$) arc (120:60:2.65) -- cycle;
      \draw[sbRed!70] ($(7,-2.6)+(60:3.35)$) arc (60:120:3.35) -- ($(7,-2.6)+(120:2.65)$) arc (120:60:2.65) -- cycle;
      \node[sbannot,text=black,anchor=west] at (-2.9,0) {\lab};
    \end{scope}
  }
  % ---- row (a): EKF ----
  \begin{scope}[yshift=0cm]
    \fill[sbBlue] (0,0) circle (1.6pt);
    \draw[sbBlue,thick] (7,0.4) ellipse (1.55 and 0.33);
    \fill[sbBlue] (7,0.4) circle (1.6pt);
    \draw[sbBlue,dashed] (5.45,0.4) -- (8.55,0.4);
    \node[sbannot,text=sbBlue,anchor=south] at (7,0.78) {tangent image of the mean};
    \draw[sbRed!80!black] (6.9,0.27) -- (7.1,0.47) (6.9,0.47) -- (7.1,0.27);
    \node[sbannot,text=sbRed!80!black,anchor=north] at (7.35,0.25) {true mean};
  \end{scope}
  % ---- row (b): UKF ----
  \begin{scope}[yshift=-1.9cm]
    \foreach \p in {(0,0),(1.2,0),(-1.2,0),(0,0.4),(0,-0.4)}{\fill[sbOrange] \p circle (1.8pt);}
    % images of the sigma points: (u,w) -> (7 + R cos(90+20u), -2.6 + R sin(90+20u)), R = 3 + 0.7 w
    \foreach \u/\w in {0/0,1.2/0,-1.2/0,0/0.4,0/-0.4}{
      \fill[sbOrange] ({7+(3+0.7*\w)*cos(90+20*\u)},{-2.6+(3+0.7*\w)*sin(90+20*\u)}) circle (1.8pt);
    }
    \draw[sbOrange,thick] (7,0.27) ellipse (1.5 and 0.38);
    \fill[sbOrange] (7,0.27) circle (1.6pt);
    \node[sbannot,text=sbOrange!80!black,anchor=south] at (7,0.75) {ellipse fitted to the images};
  \end{scope}
  % ---- row (c): PF ----
  \begin{scope}[yshift=-3.8cm]
    \pgfmathsetseed{1919}
    \foreach \i in {1,...,70}{
      \pgfmathsetmacro{\u}{0.55*(rand+rand+rand)}
      \pgfmathsetmacro{\w}{0.18*(rand+rand+rand)}
      \fill[sbPurple] (\u,\w) circle (0.8pt);
      \fill[sbPurple] ({7+(3+0.7*\w)*cos(90+20*\u)},{-2.6+(3+0.7*\w)*sin(90+20*\u)}) circle (0.8pt);
    }
    \node[sbannot,text=sbPurple,anchor=south] at (7,0.75) {the cloud is the belief};
  \end{scope}
\end{tikzpicture}

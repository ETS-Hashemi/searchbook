\section{Theorem-like environments}
\begin{definition}[Admissible heuristic]\label{def:styletest-adm}
A heuristic $\hcost$ is \textbf{admissible} if $\hcost(n) \le \hcost^*(n)$ for all $n \in V$.\index{heuristic!admissible}
\end{definition}
\begin{example}[Manhattan distance]\label{ex:styletest-manhattan}
On a 4-connected grid, $\hcost(n) = \abs{x_n - x_g} + \abs{y_n - y_g}$ is admissible.
\end{example}
\begin{theorem}[Optimality of \astar]\label{thm:styletest-opt}
If $\hcost$ is admissible, \astar returns a least-cost path.
\end{theorem}
\begin{proof}
Suppose the goal is expanded with cost $C > C^*$. Some node on an optimal path sits in \Open with $\fcost \le C^*$, contradicting the choice rule. \qedhere
\end{proof}
\begin{lemma}\label{lem:styletest}
Consistency implies admissibility.
\end{lemma}
\begin{remark}
Vectors are written $\pos$, $\vel$, matrices $\mat{A}$, transposes $\mat{A}\T$, norms $\norm{\pos - \pos'}$, sets $\set{1,\dots,k}$, expectation $\E[\state]$, covariance $\Cov(\state)$, and $\argmin_{\ctrl} J(\ctrl)$; the Gaussian is $\Normal(\vect{\mu}, \mat{\Sigma})$; complexity $\bigO{n \log n}$.
\end{remark}
\section{Pseudocode}
\begin{algorithm}[htb]
\caption{Best-first search skeleton}
\label{alg:styletest-search}
\KwIn{graph $G=(V,E)$, start $s$, goal $g$, heuristic $\hcost$}
\KwOut{a path from $s$ to $g$ or \emph{failure}}
\Fn{\StyleTestSearch{$G$, $s$, $g$}}{
  $\gcost(s) \gets 0$; \Open $\gets \set{s}$; \Closed $\gets \emptyset$\;
  \While{\Open $\neq \emptyset$}{
    $n \gets \argmin_{n' \in \text{\Open}} \fcost(n')$\label{alg:styletest-search:pop}\;
    \If{$n = g$}{\KwRet{path reconstructed from parents}\;}
    move $n$ from \Open to \Closed\;
    \ForEach{$n' \in \Succ(n)$}{
      \If{$\gcost(n) + c(n,n') < \gcost(n')$}{
        $\gcost(n') \gets \gcost(n) + c(n,n')$; parent$(n') \gets n$\tcp*{relaxation}
        insert $n'$ into \Open (or update its key)\;
      }
    }
  }
  \KwRet{failure}\;
}
\end{algorithm}
Line~\ref{alg:styletest-search:pop} of \cref{alg:styletest-search} selects the node.
\section{Listing}
\begin{lstlisting}[caption={A minimal priority-queue search.},label={lst:styletest-search}]
import heapq

def search(start, goal, succ, h):
    """Return the g-cost of goal or None."""
    g = {start: 0.0}
    frontier = [(h(start), start)]
    while frontier:
        f, n = heapq.heappop(frontier)
        if n == goal:
            return g[n]
        for m, c in succ(n):  # (neighbour, edge cost)
            if g[n] + c < g.get(m, float("inf")):
                g[m] = g[n] + c
                heapq.heappush(frontier, (g[m] + h(m), m))
    return None
\end{lstlisting}
\Cref{lst:styletest-search} uses \code{heapq}.

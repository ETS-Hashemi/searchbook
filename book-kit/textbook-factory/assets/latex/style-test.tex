\section{Theorem-like environments}
\begin{definition}[Admissible estimate]\label{def:styletest-adm}
An estimate $h$ is \textbf{admissible} if $h(n) \le h^*(n)$ for all $n$.\index{estimate!admissible}
\end{definition}
\begin{example}[A small example]\label{ex:styletest-ex}
On a grid, $h(n) = \abs{x_n - x_g} + \abs{y_n - y_g}$ is admissible.
\end{example}
\begin{theorem}[A theorem]\label{thm:styletest-opt}

\section{Pseudocode}
\begin{algorithm}[htb]
\caption{Best-first search skeleton}
\label{alg:styletest-search}
\KwIn{graph $G=(V,E)$, start $s$, goal $g$, estimate $h$}
\KwOut{a path from $s$ to $g$ or \emph{failure}}
\Fn{\StyleTestSearch{$G$, $s$, $g$}}{
  $g(s) \gets 0$; open $\gets \set{s}$\;
  \While{open $\neq \emptyset$}{
    $n \gets \argmin_{n' \in \text{open}} f(n')$\label{alg:styletest-search:pop}\;
    \If{$n = g$}{\KwRet{the path reconstructed from parents}\;}
    \ForEach{successor $n'$ of $n$}{
      update $g(n')$ and its parent\tcp*{relaxation}
    }
  }
  \KwRet{failure}\;
}
\end{algorithm}
Line~\ref{alg:styletest-search:pop} pops the best candidate.
